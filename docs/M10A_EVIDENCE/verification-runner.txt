const fs=require('node:fs'),path=require('node:path'),cp=require('node:child_process'),crypto=require('node:crypto'),os=require('node:os');
const [phase,start]=process.argv.slice(2),dir=path.resolve('docs/M10A_EVIDENCE');
const command='powershell.exe -NoProfile -ExecutionPolicy Bypass -File scripts/verify.ps1';
const hash=b=>crypto.createHash('sha256').update(b).digest('hex');
const stamp=()=>cp.execFileSync('powershell.exe',['-NoProfile','-Command','Get-Date -Format "yyyy-MM-dd HH:mm:ss K"'],{encoding:'utf8'}).trim();
const known=['docs/M10_T01_EVIDENCE/geometry.json',...['table','five-cards','four-leaves','text-fallback'].flatMap(p=>[1280,768,320].map(w=>'docs/M10_T01_EVIDENCE/'+p+'-'+w+'.png')),...['scene-open','scene-dealt'].flatMap(p=>[1280,768,320].map(w=>'docs/M10_T01_EVIDENCE/repair09/'+p+'-'+w+'.png')),...['m9-ready','m9-results','m9-tools','m9-table-1280','m9-table-768','m9-table-320'].map(p=>'docs/images/'+p+'.png')];
const before=new Map(known.map(n=>[n,fs.readFileSync(n)])),backupDir=fs.mkdtempSync(path.join(os.tmpdir(),'cbg-m10a-'+phase+'-')),originals=new Map();
for(const [n,b]of before){const p=path.join(backupDir,hash(b)+path.extname(n));fs.writeFileSync(p,b);originals.set(n,p);}
fs.mkdirSync(path.join(dir,'generated'),{recursive:true});
function preserve(b,extension){
 const digest=hash(b),old=path.resolve('docs/M10_T01_EVIDENCE/repair11/generated',digest+extension);
 if(fs.existsSync(old)&&hash(fs.readFileSync(old))===digest)return {retainedAt:path.relative(dir,old).replaceAll('\\','/'),sha256:digest,bytes:b.length,existingImmutableBytes:true};
 const target=path.join(dir,'generated',digest+extension);if(!fs.existsSync(target))fs.writeFileSync(target,b);
 if(hash(fs.readFileSync(target))!==digest)throw Error('Retained output mismatch');
 return {retainedAt:path.relative(dir,target).replaceAll('\\','/'),sha256:digest,bytes:b.length,existingImmutableBytes:false};
}
const log=path.join(dir,phase+'.log'),fd=fs.openSync(log,'wx');
console.log(JSON.stringify({phase,start,command,status:'RUNNING'}));
const result=cp.spawnSync('powershell.exe',['-NoProfile','-ExecutionPolicy','Bypass','-File','scripts/verify.ps1'],{stdio:['ignore',fd,fd]});fs.closeSync(fd);
const preliminary={start,command,exitCode:result.status,status:result.error?'BLOCKED':result.status===0?'PASS':'FAIL',error:result.error?String(result.error):null,administrativeRestoration:'PENDING'};
fs.writeFileSync(path.join(dir,phase+'.json'),JSON.stringify(preliminary,null,2)+'\n');
console.log(JSON.stringify({commandExitCode:result.status,commandStatus:preliminary.status}));
const captures=[],quote=p=>"'"+p.replaceAll("'","''")+"'";
for(const [n,b]of before){
 const current=fs.readFileSync(n);if(current.equals(b))continue;
 const retained=preserve(current,path.extname(n)),absolute=path.resolve(n),replacement=path.join(path.dirname(absolute),'.m10a-restore-'+phase+'-'+path.basename(n)),replacedBackup=path.join(backupDir,'replaced-'+path.basename(n));
 if(!absolute.startsWith(process.cwd()+path.sep)||fs.existsSync(replacement))throw Error('Unsafe or occupied restore path');
 cp.execFileSync('powershell.exe',['-NoProfile','-Command',"$ErrorActionPreference = 'Stop'\nCopy-Item -LiteralPath "+quote(originals.get(n))+" -Destination "+quote(replacement)+"\n[System.IO.File]::Replace("+quote(replacement)+","+quote(absolute)+","+quote(replacedBackup)+")"],{stdio:['ignore','pipe','pipe']});
 if(hash(fs.readFileSync(replacedBackup))!==retained.sha256||hash(fs.readFileSync(n))!==hash(b))throw Error('Known output restoration mismatch');
 captures.push({path:n,beforeSha256:hash(b),generatedSha256:retained.sha256,...retained,restoredSha256:hash(fs.readFileSync(n))});
}
const outputs=[];
function collect(root){if(!fs.existsSync(root))return;for(const e of fs.readdirSync(root,{withFileTypes:true})){const n=path.join(root,e.name);if(e.isDirectory())collect(n);else if(e.name!=='.last-run.json')outputs.push({path:n.replaceAll('\\','/'),...preserve(fs.readFileSync(n),path.extname(n))});}}
collect('test-results');
const receipt={...preliminary,end:stamp(),temporaryOriginalBackup:backupDir,administrativeRestoration:'PASS',knownWhitelistPaths:known.length,knownGeneratedCaptures:captures,testScopedOutputs:outputs,note:'Documentation-only baseline regression. No M10A visual acceptance claimed. Only25 explicitly inspected legacy harness outputs preserved/restored; canonical PA1 stays read-only. Existing identical generated bytes are referenced without duplication.'};
fs.writeFileSync(path.join(dir,phase+'.json'),JSON.stringify(receipt,null,2)+'\n');
console.log(JSON.stringify({phase,status:receipt.status,exitCode:result.status,end:receipt.end,knownCaptures:captures.length,testScopedOutputs:outputs.length}));
console.log(fs.readFileSync(log,'utf8').split(/\r?\n/).slice(-10).join('\n'));
process.exitCode=result.status??1;
