import { expect, it } from 'vitest';
import { existsSync } from 'node:fs';
import { execFileSync } from 'node:child_process';
import ts from 'typescript';

const baseline = '8326f846ad753b79fd8d35f76b00f28854e2f448';
function git(...args: string[]) {
  return execFileSync('git',['-c',`safe.directory=${process.cwd().replaceAll('\\','/')}`,...args],{encoding:'utf8'}).replaceAll('\r\n','\n').trimEnd();
}
// BEGIN PA1 historical RA1 input adapter
// Keep every RA1 delivery assertion at the accepted pre-PA1 boundary.
// Current PA1 source/assertion and inventory checks live in tests/pa1.
const ra1Sha = 'e8e8e2e1586611473f0cdb94540ce995d9bd64e7';
const historicalListings = new Map<string, string>();
let historicalTexts: Map<string, string> | null = null;
function readFileSync(file: string, encoding: 'utf8') {
  if (encoding !== 'utf8') throw new Error('Historical text input required');
  if (!historicalTexts) {
    // Same immutable inventory inputs, loaded inside the check in one Git batch.
    const paths = [...new Set([
      ...readdirSync('tests',{recursive:true}).filter(path=>/^ra1[\\/].*\.test\.tsx?$/.test(path)).map(path=>`tests/${path}`),
      ...readdirSync('tests/browser').filter(path=>path.endsWith('.spec.ts')).map(path=>`tests/browser/${path}`),
      ...docs,
    ])];
    const blobs = execFileSync('git',['-c',`safe.directory=${process.cwd().replaceAll('\\','/')}`,'cat-file','--batch'],
      { input: paths.map(path=>`${ra1Sha}:${path}\n`).join('') });
    const texts = new Map<string, string>();
    let offset = 0;
    for (const path of paths) {
      const newline = blobs.indexOf(10,offset), header = blobs.subarray(offset,newline).toString('ascii');
      if (!/^[a-f0-9]{40} blob \d+$/.test(header)) throw new Error(`Historical blob unavailable: ${path}`);
      const size = Number(header.split(' ')[2]);
      texts.set(path,blobs.subarray(newline+1,newline+1+size).toString('utf8').replaceAll('\r\n','\n').trimEnd());
      offset = newline+1+size+1;
    }
    if (offset !== blobs.length) throw new Error('Historical batch framing mismatch');
    historicalTexts = texts;
  }
  const cached = historicalTexts.get(file);
  if (cached !== undefined) return cached;
  const text = git('show',`${ra1Sha}:${file}`);
  historicalTexts.set(file,text);
  return text;
}
function readdirSync(directory: string, options?: { recursive: boolean }) {
  let listing = historicalListings.get(directory);
  if (listing === undefined) { listing = git('ls-tree','-r','--name-only',ra1Sha,directory); historicalListings.set(directory,listing); }
  return listing.split('\n')
    .map(file => file.slice(directory.length + 1)).filter(file => options?.recursive || !file.includes('/'));
}
// END PA1 historical RA1 input adapter
const docs = ['README.md',...['STATE','PLAN','DEVELOPMENT_LOG','UX_UI','LAB_MANUAL','PORTFOLIO','SPEC','DESIGN','RA1_EVIDENCE','RA1_REVIEW_HANDOFF','RA1_MAPPING'].map(name=>`docs/${name}.md`)];
function registrations(file: string, kind: 'it'|'test') {
  const titles: string[] = []; const source = ts.createSourceFile(file,readFileSync(file,'utf8'),ts.ScriptTarget.Latest,true);
  function visit(node: ts.Node) {
    if (ts.isCallExpression(node) && ts.isIdentifier(node.expression) && node.expression.text===kind) {
      expect(ts.isStringLiteral(node.arguments[0]),file).toBe(true); titles.push((node.arguments[0] as ts.StringLiteral).text);
    }
    if (ts.isCallExpression(node) && ts.isPropertyAccessExpression(node.expression)) expect(node.expression.name.text,file).not.toMatch(/^(skip|todo|only|skipIf|runIf)$/);
    ts.forEachChild(node,visit);
  }
  visit(source); return titles;
}
it('RA1 has exactly thirty unique executable RSA owners and truthful current suite/status headers', () => {
  const files = readdirSync('tests',{recursive:true}).map(String).filter(file=>/\.test\.tsx?$/.test(file));
  const rsaFiles = files.filter(file=>/^ra1[\\/]/.test(file));
  const titles = rsaFiles.flatMap(file=>registrations(`tests/${file}`,'it'));
  const ids = titles.flatMap(title=>[...title.matchAll(/\[(RSA-\d{3})\]/g)].map(m=>m[1])).sort();
  const expected = Array.from({length:30},(_,n)=>`RSA-${String(n+1).padStart(3,'0')}`); expect(ids).toEqual(expected);
  const mapping = readFileSync('docs/RA1_MAPPING.md','utf8'); expect([...mapping.matchAll(/^\| (RSA-\d+) \|/gm)].map(m=>m[1])).toEqual(expected);
  for (const match of mapping.matchAll(/\| (tests\/ra1\/[^ |]+) \|/g)) expect(existsSync(match[1])).toBe(true);
  const browser = readdirSync('tests/browser').filter(file=>file.endsWith('.spec.ts')).flatMap(file=>registrations(`tests/browser/${file}`,'test')).length;
  expect(files).toHaveLength(73); expect(titles).toHaveLength(42); expect(browser).toBe(58);
  for (const file of docs) {
    const current = readFileSync(file,'utf8').split('<!-- END CURRENT RA1 -->')[0];
    const inventory = current.match(/\*\*(\d+) Vitest files \/ (\d+) tests\*\*, \*\*(\d+) Chromium project \/ (\d+) tests\*\*/);
    expect(inventory,file).not.toBeNull(); expect(inventory!.slice(1).map(Number),file).toEqual([files.length,978+titles.length,1,browser]);
    for (const truth of ['House Rules v1.2','M1-M8 HUMAN ACCEPTED','M9 ACCEPTED: NO','RA1 ACCEPTED: NO','Deployment NOT RUN','actual model/effort NOT VERIFIED / NOT VERIFIED',
      'RA1 fresh independent review NOT RUN','8326f846ad753b79fd8d35f76b00f28854e2f448','NO FINDINGS','`2,0,2,1,1`']) expect(current,file).toContain(truth);
  }
});

    .replace('  // MODE deliberately removes the seed. Keep the reset shoe non-natural so\n  // this active-round assertion does not depend on a random opening result.\n','')

  expect(session).toBe(git('show',`${baseline}:tests/m9/session.test.ts`));
  let documentation = readFileSync('tests/m9/documentation.test.ts','utf8').replaceAll('\r\n','\n').trimEnd();
  documentation = documentation.replace("import { existsSync, readFileSync as readCurrentFile } from 'node:fs';","import { existsSync, readFileSync, readdirSync } from 'node:fs';")

  expect(documentation).toBe(git('show',`${baseline}:tests/m9/documentation.test.ts`));
  expect(readFileSync('tests/browser/m9.spec.ts','utf8').replaceAll('\r\n','\n').trimEnd()).toBe(git('show',`${baseline}:tests/browser/m9.spec.ts`));
  const allowed = ['src/domain/profile.ts','src/domain/advancedGame.ts','src/domain/behindController.ts','src/browser/controller.ts','src/ui/Actions.tsx','src/ui/DemoTools.tsx'];
  for (const file of git('diff','--name-only',baseline,ra1Sha,'--','src').split('\n').filter(Boolean)) expect(allowed).toContain(file);
});
