}
function readdirSync(directory, options) {
    return git('ls-tree', '-r', '--name-only', ra1Sha, directory).split('\n')

const docs = ['README.md', ...['STATE', 'PLAN', 'DEVELOPMENT_LOG', 'UX_UI', 'LAB_MANUAL', 'PORTFOLIO', 'SPEC', 'DESIGN', 'RA1_EVIDENCE', 'RA1_REVIEW_HANDOFF', 'RA1_MAPPING'].map(name => `docs/${name}.md`)];
function registrations(file, kind) {
    const titles = [];
    const source = profileParse(file, readFileSync(file, 'utf8'), ts.ScriptTarget.Latest, true);
    function visit(node) {
        if (ts.isCallExpression(node) && ts.isIdentifier(node.expression) && node.expression.text === kind) {
            expect(ts.isStringLiteral(node.arguments[0]), file).toBe(true);
            titles.push(node.arguments[0].text);
        }
        if (ts.isCallExpression(node) && ts.isPropertyAccessExpression(node.expression))
            expect(node.expression.name.text, file).not.toMatch(/^(skip|todo|only|skipIf|runIf)$/);
        ts.forEachChild(node, visit);
    }
    visit(source);
    return titles;
}
it('RA1 has exactly thirty unique executable RSA owners and truthful current suite/status headers', () => {
    const files = readdirSync('tests', { recursive: true }).map(String).filter(file => /\.test\.tsx?$/.test(file));
    const rsaFiles = files.filter(file => /^ra1[\\/]/.test(file));
    const titles = rsaFiles.flatMap(file => registrations(`tests/${file}`, 'it'));
    const ids = titles.flatMap(title => [...title.matchAll(/\[(RSA-\d{3})\]/g)].map(m => m[1])).sort();
    const expected = Array.from({ length: 30 }, (_, n) => `RSA-${String(n + 1).padStart(3, '0')}`);
    expect(ids).toEqual(expected);
    const mapping = readFileSync('docs/RA1_MAPPING.md', 'utf8');
    expect([...mapping.matchAll(/^\| (RSA-\d+) \|/gm)].map(m => m[1])).toEqual(expected);
    for (const match of mapping.matchAll(/\| (tests\/ra1\/[^ |]+) \|/g))
        expect(existsSync(match[1])).toBe(true);
    const browser = readdirSync('tests/browser').filter(file => file.endsWith('.spec.ts')).flatMap(file => registrations(`tests/browser/${file}`, 'test')).length;
    expect(files).toHaveLength(73);
    expect(titles).toHaveLength(42);
    expect(browser).toBe(58);
    for (const file of docs) {
        const current = readFileSync(file, 'utf8').split('<!-- END CURRENT RA1 -->')[0];
        const inventory = current.match(/\*\*(\d+) Vitest files \/ (\d+) tests\*\*, \*\*(\d+) Chromium project \/ (\d+) tests\*\*/);
        expect(inventory, file).not.toBeNull();
        expect(inventory.slice(1).map(Number), file).toEqual([files.length, 978 + titles.length, 1, browser]);
        for (const truth of ['House Rules v1.2', 'M1-M8 HUMAN ACCEPTED', 'M9 ACCEPTED: NO', 'RA1 ACCEPTED: NO', 'Deployment NOT RUN', 'actual model/effort NOT VERIFIED / NOT VERIFIED',
            'RA1 fresh independent review NOT RUN', '8326f846ad753b79fd8d35f76b00f28854e2f448', 'NO FINDINGS', '`2,0,2,1,1`'])
            expect(current, file).toContain(truth);
    }
});
