import { chromium } from '@playwright/test';
import { readFileSync, writeFileSync } from 'node:fs';
import { execFileSync } from 'node:child_process';
import { createHash } from 'node:crypto';
const browser=await chromium.launch();
const records=[];
try {
  const page=await browser.newPage();
  for(const id of ['mage_female','noble_male']) {
    const path=`docs/images/pa1-avatar-${id}.png`;
    const old=execFileSync('git',['-c','safe.directory=C:/Users/user/Documents/GitHub/casino-blackjack','show',`3c50ab4d0183cff13f2380bd60faa31583d3e988:${path}`]);
    const current=readFileSync(path);
    const pixels=await page.evaluate(async data=>{
      const images=await Promise.all(data.map(async base64=>{const img=new Image();img.src=`data:image/png;base64,${base64}`;await img.decode();return img;}));
      const canvas=document.createElement('canvas');canvas.width=images[0].naturalWidth;canvas.height=images[0].naturalHeight;
      const context=canvas.getContext('2d');const decoded=images.map(img=>{context.clearRect(0,0,canvas.width,canvas.height);context.drawImage(img,0,0);return context.getImageData(0,0,canvas.width,canvas.height).data;});
      let changed=0;const bounds=[canvas.width,canvas.height,-1,-1];
      for(let i=0;i<decoded[0].length;i+=4) if(decoded[0].slice(i,i+4).some((value,j)=>value!==decoded[1][i+j])) {changed++;const x=(i/4)%canvas.width,y=Math.floor(i/4/canvas.width);bounds[0]=Math.min(bounds[0],x);bounds[1]=Math.min(bounds[1],y);bounds[2]=Math.max(bounds[2],x);bounds[3]=Math.max(bounds[3],y);}
      return {changed,bounds,width:canvas.width,height:canvas.height};
    },[old.toString('base64'),current.toString('base64')]);
    records.push({path,oldSha256:createHash('sha256').update(old).digest('hex'),reviewSha256:createHash('sha256').update(current).digest('hex'),...pixels});
  }
  writeFileSync('docs/PA1_INDEPENDENT_REVIEW/screenshot-differences.json',JSON.stringify(records,null,2)+'\n');
} finally {await browser.close();}
