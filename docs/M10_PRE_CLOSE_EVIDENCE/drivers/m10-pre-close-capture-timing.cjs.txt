const fs=require('node:fs'),path=require('node:path');const [label,root]=process.argv.slice(2);const walk=p=>fs.readdirSync(p,{withFileTypes:true}).flatMap(e=>e.isDirectory()?walk(p+'/'+e.name):[p+'/'+e.name]);const files=walk(root).filter(p=>/\/(native-initial-timing|bounded-native-samples)\.json$/.test(p));const rows=[];for(const p of files){const r=JSON.parse(fs.readFileSync(p));const dst='docs/M10_PRE_CLOSE_EVIDENCE/'+label+'-timing-artifacts/'+path.relative(root,p);fs.mkdirSync(path.dirname(dst),{recursive:true});fs.copyFileSync(p,dst);if(r.timing)rows.push({width:r.width,height:r.height,temperature:'cold',kind:'exact original T11-P01',elapsedMs:r.timing.end-r.timing.start,destinations:r.timing.destinations.length,record:dst});else for(const [index,s]of r.samples.entries())rows.push({width:r.width,height:r.height,temperature:s.temperature,kind:'bounded PRE-CLOSE-P01',sample:index+1,elapsedMs:s.elapsed,destinations:s.timing.destinations.length,record:dst});}const result={label,mode:'FULL_MOTION/native Chromium/no-preference',files:files.length,samples:rows,budgetMs:2500,status:rows.length===11&&rows.every(r=>r.elapsedMs<=2500&&r.destinations===16)?'PASS':'FAIL'};fs.writeFileSync('docs/M10_PRE_CLOSE_EVIDENCE/'+label+'-performance-summary.json',JSON.stringify(result,null,2)+'\n');console.log(JSON.stringify({label,files:files.length,status:result.status,samples:rows.map(r=>({width:r.width,temperature:r.temperature,elapsed:r.elapsedMs}))}));if(result.status!=='PASS')process.exitCode=1;
