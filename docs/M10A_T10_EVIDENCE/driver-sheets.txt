const fs=require('node:fs'),path=require('node:path'),{chromium}=require('@playwright/test');
const [root,match,prefix]=process.argv.slice(2),images=[];
function walk(dir){for(const e of fs.readdirSync(dir,{withFileTypes:true})){const p=path.join(dir,e.name);if(e.isDirectory())walk(p);else if(e.name.endsWith('.png')&&new RegExp(match).test(e.name))images.push(p);}}
walk(root);images.sort();
(async()=>{const b=await chromium.launch();try{const p=await b.newPage({viewport:{width:1560,height:1000}});for(let n=0;n<images.length;n+=6){const group=images.slice(n,n+6);await p.setContent('<html><style>body{margin:0;background:#ddd;display:grid;grid-template-columns:repeat(3,520px);gap:0;font:18px sans-serif}figure{margin:8px;min-width:0}img{display:block;width:100%;height:auto}figcaption{overflow-wrap:anywhere;padding:8px}</style>'+group.map(f=>'<figure><figcaption>'+path.basename(f)+'</figcaption><img src="data:image/png;base64,'+fs.readFileSync(f).toString('base64')+'"></figure>').join('')+'</html>');await p.locator('img').evaluateAll(imgs=>Promise.all(imgs.map(img=>img.decode())));await p.screenshot({path:prefix+'-'+(n/6+1)+'.png',fullPage:true});}console.log(JSON.stringify({images:images.length,pages:Math.ceil(images.length/6)}));}finally{await b.close();}})();
